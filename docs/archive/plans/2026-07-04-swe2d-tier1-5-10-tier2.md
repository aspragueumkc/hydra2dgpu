```python
def apply_cell_permutation(mesh_data: dict, cell_perm: np.ndarray) -> dict:
    """Apply an RCMK cell permutation to mesh_data in place and return it.

    Reorders ``cell_nodes`` (triangle or mixed) and ``cell_face_offsets`` /
    ``cell_face_nodes`` so downstream consumers (overlay, viewer, GPKG
    persistence) use solver order.

    Parameters
    ----------
    mesh_data : dict
        Mutable mesh dict with ``cell_nodes`` and/or ``cell_face_offsets`` +
        ``cell_face_nodes``.
    cell_perm : (n_cells,) ndarray
        Permutation array: ``new[i] = old[cell_perm[i]]``.
    """
    cp = np.asarray(cell_perm)
    if cp.size == 0:
        return mesh_data

    cn = mesh_data.get("cell_nodes")
    if cn is not None and cn.size > 0:
        n_cells_mesh = cn.size // 3 if cn.ndim == 1 else cn.shape[0]
        if n_cells_mesh == cp.size:
            if cn.ndim == 1:
                reshaped = cn.reshape(-1, 3)
                mesh_data["cell_nodes"] = reshaped[cp].ravel()
            else:
                mesh_data["cell_nodes"] = cn[cp]

    cfo = mesh_data.get("cell_face_offsets")
    cfn = mesh_data.get("cell_face_nodes")
    if cfo is not None and cfn is not None:
        n_cells = len(cfo)
        if n_cells == cp.size:
            old_offsets = cfo.copy()
            new_offsets = np.zeros_like(cfo)
            for ci in range(n_cells):
                orig_ci = int(cp[ci])
                s = int(old_offsets[orig_ci])
                e = int(old_offsets[orig_ci + 1]) if ci + 1 < n_cells else len(cfn)
                new_offsets[ci + 1] = new_offsets[ci] + (e - s) if ci + 1 < len(new_offsets) else 0
            # Recompute properly:
            new_offsets = np.zeros(len(cfo), dtype=cfo.dtype)
            for ci in range(n_cells):
                orig_ci = int(cp[ci])
                s = int(old_offsets[orig_ci])
                e = int(old_offsets[orig_ci + 1]) if orig_ci + 1 < len(old_offsets) else len(cfn)
                if ci + 1 < len(new_offsets):
                    new_offsets[ci + 1] = new_offsets[ci] + (e - s)
            total = int(new_offsets[-1])
            cfn_new = np.empty(total, dtype=cfn.dtype)
            for ci in range(n_cells):
                orig_ci = int(cp[ci])
                s = int(old_offsets[orig_ci])
                e = int(old_offsets[orig_ci + 1]) if orig_ci + 1 < len(old_offsets) else len(cfn)
                ds = int(new_offsets[ci])
                cfn_new[ds:ds + (e - s)] = cfn[s:e]
            mesh_data["cell_face_offsets"] = new_offsets
            mesh_data["cell_face_nodes"] = cfn_new

    return mesh_data
```

```python
def classify_boundary_edges(
    node_x: np.ndarray, node_y: np.ndarray,
    bc_n0: np.ndarray, bc_n1: np.ndarray,
):
    """Return (edge_lengths, side_index, side_names) for boundary edges.

    side_index: 0=left, 1=right, 2=bottom, 3=top (nearest mesh boundary).
    """
    edge_len = edge_lengths(node_x, node_y, bc_n0, bc_n1)
    xmin, xmax, ymin, ymax = mesh_bounds(node_x, node_y)
    mx = 0.5 * (node_x[bc_n0] + node_x[bc_n1]) if bc_n0.size else np.empty(0)
    my = 0.5 * (node_y[bc_n0] + node_y[bc_n1]) if bc_n0.size else np.empty(0)
    if bc_n0.size:
        d = np.vstack([
            np.abs(mx - xmin), np.abs(mx - xmax),
            np.abs(my - ymin), np.abs(my - ymax),
        ])
        side_idx = np.argmin(d, axis=0)
    else:
        side_idx = np.empty(0, dtype=np.int32)
    return edge_len, side_idx, ["left", "right", "bottom", "top"]
```

```python
"""GeoTIFF export for high-perf overlay frames.

Extracted from overlay_controller so the controller no longer inlines
GDAL driver/write logic.
"""
from __future__ import annotations

import numpy as np


def export_overlay_grid_to_geotiff(
    *,
    scalar_grid: np.ndarray,
    grid_mask: np.ndarray,
    x_min: float,
    y_max: float,
    pixel_size: float,
    crs_wkt: str,
    out_path: str,
) -> None:
    """Write a scalar grid + mask to a GeoTIFF via GDAL.

    Raises RuntimeError if GDAL is unavailable or the write fails.
    """
    from osgeo import gdal

    ny, nx = scalar_grid.shape
    driver = gdal.GetDriverByName("GTiff")
    ds = driver.Create(out_path, nx, ny, 1, gdal.GDT_Float32)
    ds.SetGeoTransform((x_min, pixel_size, 0.0, y_max, 0.0, -pixel_size))
    if crs_wkt:
        ds.SetProjection(crs_wkt)
    band = ds.GetRasterBand(1)
    band.WriteArray(scalar_grid.astype(np.float32))
    band.FlushCache()
    ds = None
```
